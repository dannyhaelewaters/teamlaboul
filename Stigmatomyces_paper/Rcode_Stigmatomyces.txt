```{r}
library(ggplot2)

ggplot(summary_data, aes(x = `Suggested_genus`, y = Count, fill = `Host_family`)) +
  geom_bar(stat = "identity") +
  theme_minimal() +
  labs(title = "Occurrence of Host Families per Suggested Genus",
       x = "Suggested Genus",
       y = "Count",
       fill = "Host Family") +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))
```

```{r}
rare_host_families <- Stigmatomyces_R %>%
  group_by(Suggested_genus, Host_family) %>%
  summarise(Count = n(), .groups = 'drop') %>%
  mutate(IsRare = Count <= 2)
rare_host_families
```

```{r}
fungi_data_modified <- Stigmatomyces_R %>%
  left_join(rare_host_families, by = c('Suggested_genus', 'Host_family')) %>%
  mutate(Host_family = ifelse(IsRare, 'Others', Host_family)) %>%
  select(-Count, -IsRare) # Clean up temporary columns

# Summarize modified data
summary_data_modified <- fungi_data_modified %>%
  group_by(Suggested_genus, Host_family) %>%
  summarise(Count = n(), .groups = 'drop')

# Plot with ggplot2
library(ggplot2)
```
