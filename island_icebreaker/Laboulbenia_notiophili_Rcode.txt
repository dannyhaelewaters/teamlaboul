```{r}
# Extract the variable contributions
var_contrib <- factoextra::get_pca_var(pca_result)$contrib

# Get the top 8 contributing variables for each principal component with descriptive summaries
top_contributors_summary <- apply(var_contrib, 2, function(x) {
  # Sort contributions in decreasing order and get the top 8
  top_vars <- head(sort(x, decreasing = TRUE), 40)
  # Create a descriptive summary
  data.frame(
    #Variable = names(top_vars),
    Contribution = top_vars,
    Description = paste("The variable", names(top_vars), 
                        "contributes", round(top_vars, 2), 
                        "% to this principal component.")
  )
})

# Print the summary of the top contributors with descriptions
```

```{r}
library(emmeans)

length_boxplot <- ggplot(data_cleaned_final, aes(x=Clade, y=(Cell_II_ratio), col= Clade)) + 
  geom_boxplot()
length_boxplot

lm.model1<-lmer((Cell_II_ratio) ~ Clade+(1|Clade:Slide), data=data_cleaned_final)
summary(lm.model1)
anova(lm.model1, ddf="Satterthwaite",type=3)

#assumptions
testDispersion(lm.model1)
testUniformity(lm.model1)
testOutliers(lm.model1)
simulationOutput <- simulateResiduals(fittedModel = lm.model1, plot = F)
residuals(simulationOutput, quantileFunction = qnorm, outlierValues = c(-7,7))
plot(simulationOutput)

emmeans(lm.model1, specs = pairwise ~ Clade, type = "respons")
means <- emmeans(lm.model1, specs = pairwise ~ Clade, type = "respons")

pwpp(means, type = "response", aes = list(label = list(size = 4))) + geom_point(size = 3)
```
